import sqlite3 import re from datetime import datetime from typing import List, Tuple, Optional DB_NAME = "recruitment.db" HTML_FILE = "recruitment_ads.html" # ==================== 1. إعداد قاعدة البيانات ==================== def setup_database(): conn = sqlite3.connect(DB_NAME) cursor = conn.cursor() cursor.execute(''' CREATE TABLE IF NOT EXISTS ads ( id INTEGER PRIMARY KEY AUTOINCREMENT, source TEXT NOT NULL, content TEXT NOT NULL, category TEXT, date TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) ''') # فهرسة لتسريع البحث cursor.execute('CREATE INDEX IF NOT EXISTS idx_content ON ads(content)') # بيانات تجريبية محدّثة sample_ads = [ ("مجموعة تليجرام الاستقدام", "متوفر عاملة منزلية فلبينية للتنازل، نقل كفالة فوري، السعر مناسب."), ("موقع إلكتروني", "مطلوب استقدام عمالة صيانة ومعلمين للعمل بالخارج، تأشيرات جاهزة."), ("مجموعة واتساب", "أنا سائق خاص أبحث عن نقل كفالة وشغل في الرياض"), ("تويتر / إكس", "متوفر تنازل عائق سائق هندي بصك تنازل جاهز."), ("تليجرام", "محتاج وظيفة مدرس لغة عربية"), ("صفحة فيسبوك", "مطلوب عمال بناء وسائقين شاحنات للاستقدام الفوري من الخارج."), ("تليجرام", "عاملة منزلية إندونيسية للتنازل في الرياض - نقل كفالة"), ("واتساب", "أبغى وظيفة سائق في جدة"), ("إكس", "استقدام عمالة منزلية فلبينية - تأشيرات متوفرة الآن"), ("فيسبوك", "مطلوب استقدام سائقين خاصين من الهند"), ] cursor.execute("DELETE FROM ads") # تنظيف للاختبار cursor.executemany( "INSERT INTO ads (source, content) VALUES (?, ?)", sample_ads ) conn.commit() conn.close() print("✅ تم إعداد قاعدة البيانات بنجاح") # ==================== 2. فلترة وتصنيف متقدمة ==================== def classify_and_filter(text: str) -> Tuple[bool, Optional[str]]: """ يرجع (مقبول؟, الفئة) الفئات: تنازل | استقدام | مطلوب_عمالة | None """ if not text or not text.strip(): return False, None text_lower = text.lower().strip() # ---- كلمات الحظر (الباحثون عن عمل) ---- negative_patterns = [ r'أبحث\s*عن', r'ابحث\s*عن', r'محتاج\s*وظيفة', r'ابغى\s*وظيفة', r'أبقى\s*شغل', r'دور\s*على\s*شغل', r'أدور\s*كفيل', r'أبغى\s*وظيفة', r'عايز\s*شغل', r'محتاج\s*كفيل', r'أبحث\s*عن\s*عمل', r'ابحث\s*عن\s*وظيفة', r'أريد\s*وظيفة' ] for pattern in negative_patterns: if re.search(pattern, text_lower): return False, None # ---- كلمات القبول + التصنيف ---- # 1. التنازل transfer_keywords = [ r'تنازل', r'للتنازل', r'نقل\s*كفالة', r'صك\s*تنازل', r'متوفر.*تنازل', r'متوفرة.*تنازل' ] for pattern in transfer_keywords: if re.search(pattern, text_lower): return True, "تنازل" # 2. الاستقدام recruitment_keywords = [ r'استقدام', r'تأشيرات', r'تأشيرة', r'استقدام\s*فوري', r'عمالة\s*من\s*الخارج' ] for pattern in recruitment_keywords: if re.search(pattern, text_lower): return True, "استقدام" # 3. مطلوب عمالة (أصحاب عمل) demand_keywords = [ r'مطلوب\s*عمالة', r'مطلوب\s*استقدام', r'مطلوب\s*عمال', r'مطلوب\s*سائق', r'مطلوب\s*عاملة' ] for pattern in demand_keywords: if re.search(pattern, text_lower): return True, "مطلوب_عمالة" # كلمات عامة إيجابية general_positive = [r'متوفر', r'متوفرة'] for pattern in general_positive: if re.search(pattern, text_lower): return True, "تنازل" # افتراضي return False, None # ==================== 3. تصدير إلى HTML محسّن ==================== def export_to_html(): conn = sqlite3.connect(DB_NAME) cursor = conn.cursor() cursor.execute("SELECT id, source, content, date FROM ads ORDER BY date DESC") all_ads = cursor.fetchall() conn.close() filtered_ads = [] for ad_id, source, content, date in all_ads: accepted, category = classify_and_filter(content) if accepted: filtered_ads.append((ad_id, source, content, category, date)) # إحصائيات total = len(filtered_ads) count_transfer = sum(1 for a in filtered_ads if a[3] == "تنازل") count_recruit = sum(1 for a in filtered_ads if a[3] == "استقدام") count_demand = sum(1 for a in filtered_ads if a[3] == "مطلوب_عمالة") html_content = f""" لوحة إعلانات الاستقدام والتنازل

إعلانات الاستقدام والتنازل

لوحة مصفاة • يتم عرض العروض فقط (بدون الباحثين عن عمل)

{total}
إجمالي المطابقة
{count_transfer}
تنازل
{count_recruit}
استقدام
{count_demand}
مطلوب عمالة
""" for ad_id, source, content, category, date in filtered_ads: # تنسيق التاريخ try: dt = datetime.fromisoformat(date.replace('Z', '+00:00')) if isinstance(date, str) else date date_str = dt.strftime("%Y-%m-%d %H:%M") if hasattr(dt, 'strftime') else str(date)[:16] except: date_str = str(date)[:16] cat_class = f"cat-{category}" html_content += f""" """ html_content += """
المصدر الفئة نص الإعلان التاريخ
{source} {category.replace('_', ' ')} {content} {date_str}
لا توجد إعلانات مطابقة للبحث أو الفلتر الحالي
تم التحديث تلقائياً • نظام فلترة الاستقدام والتنازل
""" with open(HTML_FILE, "w", encoding="utf-8") as f: f.write(html_content) print(f"✅ تم إنشاء ملف HTML: {HTML_FILE}") print(f" → إجمالي الإعلانات المقبولة: {total}") print(f" → تنازل: {count_transfer} | استقدام: {count_recruit} | مطلوب عمالة: {count_demand}") # ==================== 4. إضافة إعلان جديد (اختياري) ==================== def add_ad(source: str, content: str): """إضافة إعلان جديد إلى قاعدة البيانات""" accepted, category = classify_and_filter(content) if not accepted: print(f"❌ تم رفض الإعلان (يبدو أنه من باحث عن عمل)") return False conn = sqlite3.connect(DB_NAME) cursor = conn.cursor() cursor.execute( "INSERT INTO ads (source, content, category) VALUES (?, ?, ?)", (source, content, category) ) conn.commit() conn.close() print(f"✅ تم إضافة الإعلان بنجاح (الفئة: {category})") return True # ==================== تشغيل ==================== if __name__ == "__main__": setup_database() export_to_html() # مثال على إضافة إعلان جديد: # add_ad("تليجرام", "متوفر تنازل عاملة منزلية فلبينية في جدة")
